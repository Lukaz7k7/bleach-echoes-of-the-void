#ifndef TIPOS_H

#define TIPOS_H

#define TAM_TABULEIRO 12

typedef enum{
	CLASSE_SHINIGAMI,
	CLASSE_QUINCY,
	CLASSE_HOLLOW
} ClasseDoPersonagem;

	CASA_VAZIA,
	CASA_COMBATE,
	CASA_TESOURO,
	CASA_CURA,
	CASA_ARMADILHA,
	CASA_MISSAO,
	CASA_CHEFE
} TipoDeCasa;

typedef enum{
	ACAO_ATACAR = 1,
	ACAO_DEFENDER,
	ACAO_HABILIDADE,
	ACAO_FUGIR
} AcaoCombate;

typedef struct {
	char nome[50];
	ClasseDoPersonagem classe;
	int vida_atual;
	int vida_maxima;
	int forca;
	int defesa;
	int ativa_defesa;
	int agilidade;
	int inteligencia;
	int xp;
	int nivel;
	char arma[50];
	char roupa[50];
} Personagem;

typedef struct {
	char nome[50];
	char descricao[100];
	int vida_atual;
	int vida_maxima;
	int ataque;
	int defesa;
	int nivel;
	int xp_recompensa;
} Monstro;

typedef struct {
	int visitada;
	TipoDeCasa evento;
} Casa;

typedef struct {
	Casa casas[TAM_TABULEIRO];
	int posicao_player;
	int fase_atual;
} Tabuleiro;

#endif

#include <string.h>

#include "menu.h"

#include <stdio.h>


void limpar_terminal(void) {
    printf("\033[2J\033[H");
}

int ler_inteiro(void) {
    char linha[32];
    int numero;

    if (fgets(linha, sizeof(linha), stdin) == NULL) {
        return -1;
    }

    if (sscanf(linha, "%d", &numero) != 1) {
        return -1;
    }

    return numero;
}

	int classe = 4;
	do{

		printf(" Especialistas em velocidade e inteligencia.\n");
		printf("Vida: 90 | Forca: 7 | Defesa: 6 | Agilidade: 12 | Inteligencia: 10\n");

		printf("\n[3] HOLLOW\n");
		printf("Criaturas espirituais predadoras. Possuem");

		printf("Vida: 100 | Forca: 13 | Defesa: 4 | Agilidade: 8 | Inteligencia: 4\n");

		printf("\n");
		printf("\n");

			printf("\n");

			printf("press ENTER para continuar");

			char descarte[32];
			fgets(descarte,sizeof(descarte), stdin);
		}

	}while (classe < 1 || classe > 3);

	return classe - 1;

}

void ler_nome(char nome[], int tamanho){
	do{
		limpar_terminal();
		printf("digite o nome do seu personagem : ");
		fgets(nome , tamanho , stdin);
		nome[strcspn(nome, "\n")] = '\0';

		if(nome[0] == '\0'){
			printf("NOME INVALIDO, DIGITE NOVAMENTE !\n");
			printf("press ENTER para continuar");
			char descarte[32];
			fgets(descarte, sizeof(descarte), stdin);

		}


	}while(nome[0] == '\0');



}

#include <string.h>

#include "personagem.h"

#include <stdio.h>

void criar_personagem(Personagem *p, ClasseDoPersonagem classe, char nome[]) {
	strcpy(p->nome, nome);
	p->classe = classe;
	p->nivel = 1;
	p->xp = 0;
	p->ativa_defesa = 0;
	switch (classe) {
		case CLASSE_SHINIGAMI:
			p->vida_atual = 110;
			p->vida_maxima = 110;
			p->forca = 11;
			p->defesa = 9;
			p->agilidade = 7;
			p->inteligencia = 7;
			strcpy(p->arma, "zanpakuto");
			strcpy(p->roupa, "shihakusho");
			break;

		case CLASSE_QUINCY:
			p->vida_atual = 90;
			p->vida_maxima = 90;
			p->forca = 7;
			p->defesa = 6;
			p->agilidade = 12;
			p->inteligencia = 10;
			strcpy(p->arma, "arco espiritual");
			strcpy(p->roupa, "tunicas quincy");
			break;

		case CLASSE_HOLLOW:
			p->vida_atual = 100;
			p->vida_maxima = 100;
			p->forca = 13;
			p->defesa = 4;
			p->agilidade = 8;
			p->inteligencia = 4;
			strcpy(p->arma, "nenhuma");
			strcpy(p->roupa, "mascara hollow");
			break;
	}
}

const char *nome_da_classe(ClasseDoPersonagem classe){
	switch (classe){
		case CLASSE_SHINIGAMI:
			return "Shinigami";

		case CLASSE_QUINCY:
			return "Quincy";

		case CLASSE_HOLLOW:
			return "Hollow";
	}
	return "desconhecida";

}



void mostrar_personagem(Personagem *p) {

    printf("\n");
    printf("  +------------------------------------------------------+\n");
    printf("  |                  STATUS DO GUERREIRO                 \n");
    printf("  +------------------------------------------------------+\n");

    printf("  | NOME: %-20s CLASSE: %-14s \n",
           p->nome, nome_da_classe(p->classe));

    printf("  +------------------------------------------------------+\n");
    printf("  | VIDA: %d/%d                                      \n",
           p->vida_atual, p->vida_maxima);

    printf("  | NIVEL: %-4d XP: %-10d                       \n",
           p->nivel, p->xp);

    printf("  +------------------------------------------------------+\n");
    printf("  |                     ATRIBUTOS                        \n");

    printf("  | FORCA: %-4d DEFESA: %-4d AGILIDADE: %-4d      \n",
           p->forca, p->defesa, p->agilidade);

    printf("  | INTELIGENCIA: %-5d                                \n",
           p->inteligencia);

    printf("  +------------------------------------------------------+\n");
    printf("  |                    EQUIPAMENTOS                      \n");

    printf("  | ARMA: %-47s \n", p->arma);
    printf("  | ROUPA: %-46s \n", p->roupa);

    printf("  +------------------------------------------------------+\n");
    printf("\n");
}
